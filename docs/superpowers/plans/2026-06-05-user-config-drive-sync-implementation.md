# User Config Drive Sync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add automatic and manual synchronization of HHHL Chat account-level settings through hhhl.com Drive at `telegram-bot-chat/settings.json`.

**Architecture:** Add focused settings sync modules under `chat/src/settings/`: one schema module, one Drive wrapper, and one sync coordinator. Keep `settingsStore` responsible for local state, sync status, debounce/serialization, and UI-facing actions. Trigger sync after authorization and expose sync state plus manual save in the settings view.

**Tech Stack:** Vue 3, Pinia, TypeScript, Vite, Vitest, MSW-style fakes, hhhl Drive JSON endpoints, multipart `drive/files/create`.

## Scope Check

This is one subsystem: user settings sync for the existing `chat/` Mini App. It touches Drive API contracts, settings sync logic, settings store state, login integration, settings UI, and tests.
```ts
import { normalizeLocale, type Locale } from '@/i18n/locales';

export const CLOUD_SETTINGS_SCHEMA_VERSION = 1;
export const CLOUD_SETTINGS_APP = 'hhhl-chat';
export const SETTINGS_UPDATED_AT_KEY = 'hhhl-chat:settings-updated-at';

export type ThemeMode = 'system' | 'light' | 'dark';

export interface SettingsPreferences {
  language: Locale;
  themeMode: ThemeMode;
  favoriteUserIds: string[];
}

export interface CloudSettingsDocument {
  schemaVersion: typeof CLOUD_SETTINGS_SCHEMA_VERSION;
  app: typeof CLOUD_SETTINGS_APP;
  updatedAt: string;
  preferences: SettingsPreferences;
  [key: string]: unknown;
}

export type ParseCloudSettingsResult =
  | { ok: true; document: CloudSettingsDocument }
  | { ok: false; code: 'INVALID_JSON' | 'INVALID_SHAPE' | 'UNSUPPORTED_SCHEMA' | 'INVALID_APP' | 'INVALID_UPDATED_AT' | 'INVALID_PREFERENCES'; message: string };

function recordFrom(value: unknown): Record<string, unknown> | null {
  return value != null && typeof value === 'object' && !Array.isArray(value) ? value as Record<string, unknown> : null;
}

export function isThemeMode(value: unknown): value is ThemeMode {
  return value === 'system' || value === 'light' || value === 'dark';
}

export function normalizeFavoriteUserIds(value: unknown): string[] | null {
  if (!Array.isArray(value)) {
    return null;
  }

  return [...new Set(value
    .filter((item): item is string => typeof item === 'string')
    .map((item) => item.trim())
    .filter((item) => item !== ''))];
}

export function isUtcIsoTimestamp(value: unknown): value is string {
  if (typeof value !== 'string') {
    return false;
  }

  const timestamp = Date.parse(value);
  return Number.isFinite(timestamp) && new Date(timestamp).toISOString() === value;
}

export function compareUpdatedAt(left: string, right: string): -1 | 0 | 1 {
  if (!isUtcIsoTimestamp(left) || !isUtcIsoTimestamp(right)) {
    throw new Error('Invalid updatedAt timestamp');
  }

  const leftTime = Date.parse(left);
  const rightTime = Date.parse(right);

  if (leftTime === rightTime) {
    return 0;
  }

  return leftTime > rightTime ? 1 : -1;
}

function parsePreferences(value: unknown): SettingsPreferences | null {
  const raw = recordFrom(value);
  if (raw == null) {
    return null;
  }

  const language = normalizeLocale(typeof raw.language === 'string' ? raw.language : null);
  const themeMode = isThemeMode(raw.themeMode) ? raw.themeMode : null;
  const favoriteUserIds = normalizeFavoriteUserIds(raw.favoriteUserIds);

  if (language == null || themeMode == null || favoriteUserIds == null) {
    return null;
  }

  return {
    language,
    themeMode,
    favoriteUserIds,
  };
}

export function parseCloudSettingsJson(value: string): ParseCloudSettingsResult {
  let parsed: unknown;
  try {
    parsed = JSON.parse(value);
  } catch {
    return { ok: false, code: 'INVALID_JSON', message: 'Invalid settings JSON' };
  }

  const raw = recordFrom(parsed);
  if (raw == null) {
    return { ok: false, code: 'INVALID_SHAPE', message: 'Invalid settings document shape' };
  }

  if (raw.schemaVersion !== CLOUD_SETTINGS_SCHEMA_VERSION) {
    return { ok: false, code: 'UNSUPPORTED_SCHEMA', message: 'Unsupported settings schema version' };
  }

  if (raw.app !== CLOUD_SETTINGS_APP) {
    return { ok: false, code: 'INVALID_APP', message: 'Invalid settings app id' };
  }

  if (!isUtcIsoTimestamp(raw.updatedAt)) {
    return { ok: false, code: 'INVALID_UPDATED_AT', message: 'Invalid settings updatedAt timestamp' };
  }

  const preferences = parsePreferences(raw.preferences);
  if (preferences == null) {
    return { ok: false, code: 'INVALID_PREFERENCES', message: 'Invalid settings preferences' };
  }

  return {
    ok: true,
    document: {
      ...raw,
      schemaVersion: CLOUD_SETTINGS_SCHEMA_VERSION,
      app: CLOUD_SETTINGS_APP,
      updatedAt: raw.updatedAt,
      preferences,
    },
  };
}

export function createCloudSettingsDocument(
  preferences: SettingsPreferences,
  updatedAt: string,
  base?: CloudSettingsDocument | null,
): CloudSettingsDocument {
  if (!isUtcIsoTimestamp(updatedAt)) {
    throw new Error('Invalid updatedAt timestamp');
  }

  const preserved = base == null
    ? {}
    : Object.fromEntries(Object.entries(base).filter(([key]) => !['schemaVersion', 'app', 'updatedAt', 'preferences'].includes(key)));

  return {
    ...preserved,
    schemaVersion: CLOUD_SETTINGS_SCHEMA_VERSION,
    app: CLOUD_SETTINGS_APP,
    updatedAt,
    preferences: {
      language: preferences.language,
      themeMode: preferences.themeMode,
      favoriteUserIds: normalizeFavoriteUserIds(preferences.favoriteUserIds) ?? [],
    },
  };
}
```

```ts
import { ApiError, NetworkError, redactSensitiveText } from '@/shared/errors';
import type { DriveFile } from '@/shared/types';

export interface DriveFolderSummary {
  id: string;
  name: string;
  parentId?: string | null;
}

export interface SettingsDriveFile {
  id: string;
  name: string;
  url?: string | null;
  createdAt?: string | null;
  updatedAt?: string | null;
}

export interface SettingsDriveApiClient extends EndpointCaller {
  uploadFile: (formData: FormData) => Promise<DriveFile>;
  tokenProvider: () => string | null | undefined;
}

export interface SettingsDriveApiOptions {
  callEndpoint: SettingsDriveApiClient['callEndpoint'];
  uploadFile: SettingsDriveApiClient['uploadFile'];
  tokenProvider: SettingsDriveApiClient['tokenProvider'];
  fetchImpl?: typeof fetch;
}

function recordFrom(value: unknown): Record<string, unknown> | null {
  return value != null && typeof value === 'object' && !Array.isArray(value) ? value as Record<string, unknown> : null;
}

function stringField(value: unknown): string | null {
  return typeof value === 'string' && value.trim() !== '' ? value.trim() : null;
}

function arrayFrom(value: unknown, keys: string[]): unknown[] {
  if (Array.isArray(value)) {
    return value;
  }

  const raw = recordFrom(value);
  if (raw == null) {
    return value == null ? [] : [value];
  }

  for (const key of keys) {
    if (Array.isArray(raw[key])) {
      return raw[key];
    }
  }

  return [value];
}

function normalizeFolder(value: unknown): DriveFolderSummary | null {
  const container = recordFrom(value);
  const raw = recordFrom(container?.folder) ?? container;
  if (raw == null) {
    return null;
  }

  const id = stringField(raw.id) ?? stringField(raw.folderId) ?? stringField(raw._id);
  const name = stringField(raw.name);
  if (id == null || name == null) {
    return null;
  }

  return {
    id,
    name,
    parentId: stringField(raw.parentId),
  };
}

function normalizeFile(value: unknown): SettingsDriveFile | null {
  const container = recordFrom(value);
  const raw = recordFrom(container?.file) ?? container;
  if (raw == null) {
    return null;
  }

  const id = stringField(raw.id) ?? stringField(raw.fileId) ?? stringField(raw._id);
  const name = stringField(raw.name);
  if (id == null || name == null) {
    return null;
  }

  const rawUrl = stringField(raw.url);
  const url = rawUrl == null ? null : new URL(rawUrl, DC_HHHL_ORIGIN).toString();

  return {
    id,
    name,
    url,
    createdAt: stringField(raw.createdAt),
    updatedAt: stringField(raw.updatedAt),
  };
}

function defaultFetch(input: string | URL | Request, init?: RequestInit): Promise<Response> {
  return window.fetch(input, init);
}

export function createSettingsDriveApi(options: SettingsDriveApiOptions) {
  const fetchImpl = options.fetchImpl ?? defaultFetch;

  async function fetchJsonFile(fileUrl: string): Promise<unknown> {
    let url: URL;
    try {
      url = new URL(fileUrl, DC_HHHL_ORIGIN);
    } catch {
      throw new ApiError('DRIVE_FILE_URL_INVALID', 'Invalid Drive file URL');
    }

    if (url.origin !== DC_HHHL_ORIGIN) {
      throw new ApiError('DRIVE_FILE_URL_NOT_ALLOWED', 'Drive file URL is not allowed');
    }

    try {
      const response = await fetchImpl(url.toString());
      if (!response.ok) {
        throw new ApiError(`HTTP_${response.status}`, response.statusText, response.status);
      }

      return await response.json();
    } catch (error) {
      if (error instanceof ApiError) {
        throw error;
      }

      throw new NetworkError('NETWORK_ERROR', redactSensitiveText(error instanceof Error ? error.message : String(error)));
    }
  }

  return {
    findFolder: async (name: string, parentId?: string | null): Promise<DriveFolderSummary | null> => {
      const response = await options.callEndpoint<unknown>('drive/folders/find', parentId == null ? { name } : { name, parentId });
      return normalizeFolder(response);
    },

    createFolder: async (name: string, parentId?: string | null): Promise<DriveFolderSummary> => {
      const response = await options.callEndpoint<unknown>('drive/folders/create', parentId == null ? { name } : { name, parentId });
      const folder = normalizeFolder(response);
      if (folder == null) {
        throw new ApiError('DRIVE_FOLDER_INVALID', 'Drive folder response was invalid');
      }

      return folder;
    },

    findFiles: async (name: string, folderId: string): Promise<SettingsDriveFile[]> => {
      const response = await options.callEndpoint<unknown>('drive/files/find', { name, folderId });
      return arrayFrom(response, ['files', 'items']).map(normalizeFile).filter((file): file is SettingsDriveFile => file != null);
    },

    showFile: async (fileId: string): Promise<SettingsDriveFile> => {
      const response = await options.callEndpoint<unknown>('drive/files/show', { fileId });
      const file = normalizeFile(response);
      if (file == null) {
        throw new ApiError('DRIVE_FILE_INVALID', 'Drive file response was invalid');
      }

      return file;
    },

    fetchJsonFile,

    createJsonFile: async (folderId: string, name: string, data: unknown): Promise<SettingsDriveFile> => {
      const formData = new FormData();
      const token = options.tokenProvider();

      if (token != null && token !== '') {
        formData.set('i', token);
      }

      formData.set('folderId', folderId);
      formData.set('force', 'true');
      formData.set('isSensitive', 'false');
      formData.set('name', name);
      formData.set('file', new File([`${JSON.stringify(data)}\n`], name, { type: 'application/json' }));

      const uploaded = await options.uploadFile(formData);
      const file = normalizeFile(uploaded);
      if (file == null) {
        throw new ApiError('DRIVE_FILE_INVALID', 'Drive file response was invalid');
      }

      return file;
    },

    deleteFile: (fileId: string): Promise<unknown> => options.callEndpoint('drive/files/delete', { fileId }),
  };
}

export type SettingsDriveApi = ReturnType<typeof createSettingsDriveApi>;
```

```ts
import {
  compareUpdatedAt,
  createCloudSettingsDocument,
  parseCloudSettingsJson,
  type CloudSettingsDocument,
  type SettingsPreferences,
} from './settingsConfig';
import type { SettingsDriveApi, SettingsDriveFile } from './settingsDriveApi';

export const SETTINGS_SYNC_FOLDER_NAME = 'telegram-bot-chat';
export const SETTINGS_SYNC_FILE_NAME = 'settings.json';

export interface LocalSettingsSnapshot {
  preferences: SettingsPreferences;
  updatedAt: string;
  baseDocument?: CloudSettingsDocument | null;
}

export type SettingsSyncStatus = 'created' | 'loaded-cloud' | 'saved-local' | 'unchanged';

export interface SettingsSyncResult {
  status: SettingsSyncStatus;
  document: CloudSettingsDocument;
  fileId?: string;
  syncedAt: string;
}

export interface SettingsSyncServiceOptions {
  drive: SettingsDriveApi;
  now?: () => Date;
}

interface CloudCandidate {
  file: SettingsDriveFile;
  document: CloudSettingsDocument;
}

function nowIso(now: () => Date): string {
  return now().toISOString();
}

function stringifyForParser(value: unknown): string {
  return typeof value === 'string' ? value : JSON.stringify(value) ?? '';
}

function newestCandidate(candidates: CloudCandidate[]): CloudCandidate {
  return [...candidates].sort((left, right) => compareUpdatedAt(right.document.updatedAt, left.document.updatedAt))[0];
}

export function createSettingsSyncService(options: SettingsSyncServiceOptions) {
  const now = options.now ?? (() => new Date());

  async function ensureFolder() {
    const existing = await options.drive.findFolder(SETTINGS_SYNC_FOLDER_NAME);
    if (existing != null) {
      return existing;
    }

    return options.drive.createFolder(SETTINGS_SYNC_FOLDER_NAME);
  }

  async function readCandidates(files: SettingsDriveFile[]): Promise<CloudCandidate[]> {
    const candidates: CloudCandidate[] = [];

    for (const file of files.filter((item) => item.name === SETTINGS_SYNC_FILE_NAME)) {
      const shownFile = await options.drive.showFile(file.id);
      if (shownFile.url == null || shownFile.url === '') {
        continue;
      }

      try {
        const json = await options.drive.fetchJsonFile(shownFile.url);
        const parsed = parseCloudSettingsJson(stringifyForParser(json));
        if (parsed.ok) {
          candidates.push({ file: shownFile, document: parsed.document });
        }
      } catch {
        // A bad duplicate must not hide another valid settings file.
      }
    }

    return candidates;
  }

  async function readCloud(folderId: string): Promise<{ files: SettingsDriveFile[]; candidates: CloudCandidate[] }> {
    const files = await options.drive.findFiles(SETTINGS_SYNC_FILE_NAME, folderId);
    const exactFiles = files.filter((file) => file.name === SETTINGS_SYNC_FILE_NAME);
    const candidates = await readCandidates(exactFiles);
    return { files: exactFiles, candidates };
  }

  async function createCloud(folderId: string, snapshot: LocalSettingsSnapshot, status: 'created' | 'saved-local'): Promise<SettingsSyncResult> {
    const document = createCloudSettingsDocument(snapshot.preferences, snapshot.updatedAt, snapshot.baseDocument ?? null);
    const created = await options.drive.createJsonFile(folderId, SETTINGS_SYNC_FILE_NAME, document);

    return {
      status,
      document,
      fileId: created.id,
      syncedAt: nowIso(now),
    };
  }

  async function replaceCloud(folderId: string, files: SettingsDriveFile[], snapshot: LocalSettingsSnapshot): Promise<SettingsSyncResult> {
    for (const file of files) {
      await options.drive.deleteFile(file.id);
    }

    return createCloud(folderId, snapshot, 'saved-local');
  }

  async function reconcile(snapshot: LocalSettingsSnapshot, createStatus: 'created' | 'saved-local'): Promise<SettingsSyncResult> {
    const folder = await ensureFolder();
    const { files, candidates } = await readCloud(folder.id);

    if (files.length === 0) {
      return createCloud(folder.id, snapshot, createStatus);
    }

    if (candidates.length === 0) {
      throw new Error('No valid cloud settings document found');
    }

    const selected = newestCandidate(candidates);
    const comparison = compareUpdatedAt(snapshot.updatedAt, selected.document.updatedAt);

    if (comparison < 0) {
      return {
        status: 'loaded-cloud',
        document: selected.document,
        fileId: selected.file.id,
        syncedAt: nowIso(now),
      };
    }

    if (comparison === 0 && files.length === 1) {
      return {
        status: 'unchanged',
        document: selected.document,
        fileId: selected.file.id,
        syncedAt: nowIso(now),
      };
    }

    return replaceCloud(folder.id, files, snapshot);
  }

  return {
    syncAfterLogin: (snapshot: LocalSettingsSnapshot): Promise<SettingsSyncResult> => reconcile(snapshot, 'created'),
    save: (snapshot: LocalSettingsSnapshot): Promise<SettingsSyncResult> => reconcile(snapshot, 'saved-local'),
  };
}

export type SettingsSyncService = ReturnType<typeof createSettingsSyncService>;
```

```ts
    applyCloudDocument(document: CloudSettingsDocument, storage: LocalStorageAdapter) {
      this.language = document.preferences.language;
      this.themeMode = document.preferences.themeMode;
      this.favoriteUserIds = normalizeFavoriteUserIds(document.preferences.favoriteUserIds) ?? [];
      this.localUpdatedAt = document.updatedAt;

    },

    applySyncResult(result: SettingsSyncResult, storage: LocalStorageAdapter) {
      if (result.status === 'loaded-cloud') {
        this.applyCloudDocument(result.document, storage);
      } else {
        this.baseCloudDocument = result.document;
      }

      this.syncStatus = 'synced';
      this.syncError = null;
      this.lastSyncedAt = result.syncedAt;
    },

    async syncAfterLogin(input?: SettingsDependencyInput) {
      const dependencies = resolveDependencies(input);
      if (dependencies.sync == null) {
        return;
      }

      if (this.localUpdatedAt == null) {
        this.init(dependencies);
      }

      this.syncStatus = 'loading';
      this.syncError = null;

      try {
        const result = await dependencies.sync.syncAfterLogin(this.localSnapshot());
        this.applySyncResult(result, dependencies.storage);
      } catch (error) {
        this.syncStatus = 'failed';
        this.syncError = redactSensitiveText(error instanceof Error ? error.message : String(error));
      }
    },

    async saveToCloud(input?: SettingsDependencyInput) {
      const dependencies = resolveDependencies(input);
      if (dependencies.sync == null) {
        return;
      }

      if (this.localUpdatedAt == null) {
        this.init(dependencies);
      }

      this.syncStatus = 'saving';
      this.syncError = null;

      try {
        const result = await dependencies.sync.save(this.localSnapshot());
        this.applySyncResult(result, dependencies.storage);
      } catch (error) {
        this.syncStatus = 'failed';
        this.syncError = redactSensitiveText(error instanceof Error ? error.message : String(error));
      }
    },

    queueAutoSave(input?: SettingsDependencyInput) {
      const dependencies = resolveDependencies(input);
      if (dependencies.sync == null) {
        return;
      }

      if (this.autoSaveTimer != null) {
        window.clearTimeout(this.autoSaveTimer);
      }

      this.autoSaveTimer = window.setTimeout(() => {
        this.autoSaveTimer = null;
        void this.runQueuedAutoSave(dependencies);
      }, dependencies.debounceMs);
    },

    async runQueuedAutoSave(dependencies: SettingsStoreDependencies) {
      if (dependencies.sync == null) {
        return;
      }

      if (this.autoSaveInFlight) {
        this.autoSaveQueued = true;
        return;
      }

      this.autoSaveInFlight = true;
      await this.saveToCloud(dependencies);
      this.autoSaveInFlight = false;

      if (this.autoSaveQueued) {
        this.autoSaveQueued = false;
        await this.runQueuedAutoSave(dependencies);
      }
    },
```
