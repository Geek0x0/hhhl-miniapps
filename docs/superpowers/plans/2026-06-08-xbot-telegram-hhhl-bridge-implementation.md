# XBot Telegram HHHL Bridge Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a new `xbot/` Cloudflare Worker subproject that bridges one authorized Telegram private chat with one bound HHHL room using KV for state and a Durable Object for HHHL WebSocket forwarding.

**Architecture:** `xbot` is a separate Worker from the existing Mini App `bot`. The Worker handles Telegram webhooks, authorization, commands, and Telegram-to-HHHL sends; KV stores binding/status/message mappings; a per-Telegram-user Durable Object owns HHHL WebSocket connection, backfill, dedupe, and HHHL-to-Telegram forwarding.

**Tech Stack:** TypeScript, Cloudflare Workers, Cloudflare KV, Durable Objects, Wrangler, Vitest, Telegram Bot API, HHHL API, HHHL streaming WebSocket.

---

## Scope Check

This plan implements the approved single subsystem from [the design spec](../specs/2026-06-08-xbot-telegram-hhhl-bridge-design.md): the `xbot` Telegram-to-HHHL bridge. It does not change `chat/` UI and does not extend the existing `bot/` Mini App entrypoint Worker.

```ts
import type { AppConfig, ConfigResult, Env } from './env';

const DEFAULT_HHHL_ORIGIN = 'https://dc.hhhl.cc';

function nonEmpty(value: string | undefined): string | null {
  return typeof value === 'string' && value.trim() !== '' ? value.trim() : null;
}

function positiveInteger(name: string, value: string | undefined, fallback: number): number | { error: string } {
  if (value == null || value.trim() === '') {
    return fallback;
  }

  const parsed = Number(value);
  if (!Number.isInteger(parsed) || parsed <= 0) {
    return { error: `${name} must be a positive integer` };
  }

  return parsed;
}

export function readConfig(env: Env): ConfigResult {
  const missing: string[] = [];
  const botToken = nonEmpty(env.BOT_TOKEN);
  const hhhlToken = nonEmpty(env.HHHL_TOKEN);
  const allowedTelegramUserId = nonEmpty(env.ALLOWED_TELEGRAM_USER_ID);

  if (botToken == null) missing.push('BOT_TOKEN');
  if (hhhlToken == null) missing.push('HHHL_TOKEN');
  if (allowedTelegramUserId == null) missing.push('ALLOWED_TELEGRAM_USER_ID');

  if (missing.length > 0) {
    return { ok: false, error: `missing ${missing.join(', ')}` };
  }

  const initialHistoryLimit = positiveInteger('INITIAL_HISTORY_LIMIT', env.INITIAL_HISTORY_LIMIT, 30);
  if (typeof initialHistoryLimit !== 'number') return { ok: false, error: initialHistoryLimit.error };

  const reconnectBaseDelayMs = positiveInteger('RECONNECT_BASE_DELAY_MS', env.RECONNECT_BASE_DELAY_MS, 1000);
  if (typeof reconnectBaseDelayMs !== 'number') return { ok: false, error: reconnectBaseDelayMs.error };

  const reconnectMaxDelayMs = positiveInteger('RECONNECT_MAX_DELAY_MS', env.RECONNECT_MAX_DELAY_MS, 60000);
  if (typeof reconnectMaxDelayMs !== 'number') return { ok: false, error: reconnectMaxDelayMs.error };

  const hhhlOrigin = nonEmpty(env.HHHL_ORIGIN) ?? DEFAULT_HHHL_ORIGIN;
  const hhhlApiBaseUrl = nonEmpty(env.HHHL_API_BASE_URL) ?? `${hhhlOrigin.replace(/\/$/, '')}/api`;
  const kvKeyPrefix = nonEmpty(env.KV_KEY_PREFIX) ?? 'xbot';

  const value: AppConfig = {
    botToken: botToken as string,
    hhhlToken: hhhlToken as string,
    allowedTelegramUserId: allowedTelegramUserId as string,
    hhhlOrigin,
    hhhlApiBaseUrl,
    initialHistoryLimit,
    reconnectBaseDelayMs,
    reconnectMaxDelayMs,
    kvKeyPrefix,
  };

  return { ok: true, value };
}
```

const TOKEN_FIELD_PATTERN =
  /\b((?:BOT_TOKEN|HHHL_TOKEN|botToken|hhhlToken|token|access_token|i)\s*[=:]\s*["']?)(?!\[redacted\])[^&\s"',;)}]+(["']?)/gi;

function escapeRegExp(value: string): string {
  return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
}

export function redactSensitiveText(value: string, secrets: string[] = []): string {
  let redacted = value.replace(TOKEN_FIELD_PATTERN, '$1[redacted]$2');

  for (const secret of secrets) {
    if (secret.trim() === '') continue;
    redacted = redacted.replace(new RegExp(escapeRegExp(secret), 'g'), '[redacted]');
  }

  return redacted;
}

export interface StateKeys {
  binding: (telegramUserId: string) => string;
  telegramMap: (telegramUserId: string, telegramMessageId: number) => string;
  hhhlMap: (roomId: string, hhhlMessageId: string) => string;

  status: (telegramUserId: string) => string;

function encodePart(value: string | number): string {
  return encodeURIComponent(String(value));
}

export function createKeys(prefix: string): StateKeys {
  const root = prefix.replace(/:+$/, '');

  return {
    binding: (telegramUserId) => `${root}:binding:${encodePart(telegramUserId)}`,
    telegramMap: (telegramUserId, telegramMessageId) => `${root}:map:telegram:${encodePart(telegramUserId)}:${encodePart(telegramMessageId)}`,
    hhhlMap: (roomId, hhhlMessageId) => `${root}:map:hhhl:${encodePart(roomId)}:${encodePart(hhhlMessageId)}`,
    telegramMapPrefix: (telegramUserId) => `${root}:map:telegram:${encodePart(telegramUserId)}:`,
    hhhlMapPrefix: (roomId) => `${root}:map:hhhl:${encodePart(roomId)}:`,
    status: (telegramUserId) => `${root}:status:${encodePart(telegramUserId)}`,

export interface BindingState {
  version: 1;
  telegramUserId: string;
  roomId: string;
  roomName: string;
  boundAt: string;
  lastSeenMessageId: string | null;
}

export interface MessageMapState {
  version: 1;
  roomId: string;
  hhhlMessageId: string;
  telegramUserId: string;
  telegramMessageId: number;
  createdAt: string;
}

export interface RealtimeStatusState {
  version: 1;
  state: 'stopped' | 'connecting' | 'connected' | 'backing_off';
  connectedAt: string | null;
  lastError: string | null;
  nextReconnectAt: string | null;
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function stringField(value: unknown): string | null {
  return typeof value === 'string' && value.trim() !== '' ? value : null;
}

export function parseBindingState(value: unknown): BindingState | null {
  if (!isRecord(value) || value.version !== 1) return null;
  const telegramUserId = stringField(value.telegramUserId);
  const roomId = stringField(value.roomId);
  const roomName = stringField(value.roomName);
  const boundAt = stringField(value.boundAt);
  const lastSeenMessageId = value.lastSeenMessageId === null ? null : stringField(value.lastSeenMessageId);

  if (telegramUserId == null || roomId == null || roomName == null || boundAt == null) return null;

  return { version: 1, telegramUserId, roomId, roomName, boundAt, lastSeenMessageId };
}

export function parseMessageMapState(value: unknown): MessageMapState | null {
  if (!isRecord(value) || value.version !== 1) return null;
  const roomId = stringField(value.roomId);
  const hhhlMessageId = stringField(value.hhhlMessageId);
  const telegramUserId = stringField(value.telegramUserId);
  const createdAt = stringField(value.createdAt);
  const telegramMessageId = typeof value.telegramMessageId === 'number' && Number.isFinite(value.telegramMessageId) ? value.telegramMessageId : null;

  if (roomId == null || hhhlMessageId == null || telegramUserId == null || telegramMessageId == null || createdAt == null) return null;

  return { version: 1, roomId, hhhlMessageId, telegramUserId, telegramMessageId, createdAt };
}

export function parseRealtimeStatusState(value: unknown): RealtimeStatusState | null {
  if (!isRecord(value) || value.version !== 1) return null;
  const state = value.state;
  if (state !== 'stopped' && state !== 'connecting' && state !== 'connected' && state !== 'backing_off') return null;

  return {

  '可用命令：',
  '/bind <roomId> [显示名] - 绑定一个 HHHL 聊天室',
  '/unbind - 解绑当前聊天室并停止转发',
  '/rename <显示名> - 修改当前聊天室显示名',
  '/list - 查看当前绑定',
  '/status - 查看实时连接状态',
  '/help - 显示帮助',
].join('\n');

export function parseCommand(text: string): BotCommand | null {
  const trimmed = text.trim();
  if (!trimmed.startsWith('/')) return null;

  const [rawCommand, ...parts] = trimmed.split(/\s+/);
  const command = rawCommand.replace(/@\w+$/, '').toLowerCase();

  if (command === '/bind') {
    const roomId = parts[0];
    if (roomId == null || roomId.trim() === '') {
      return { type: 'invalid', reason: '用法：/bind <roomId> [显示名]' };
    }

    const roomName = parts.slice(1).join(' ').trim();
    return { type: 'bind', roomId, roomName: roomName === '' ? null : roomName };
  }

  if (command === '/unbind') return { type: 'unbind' };
  if (command === '/list') return { type: 'list' };
  if (command === '/status') return { type: 'status' };
  if (command === '/help' || command === '/start') return { type: 'help' };

  if (command === '/rename') {
    const roomName = parts.join(' ').trim();
    if (roomName === '') {
      return { type: 'invalid', reason: '用法：/rename <显示名>' };
    }
    return { type: 'rename', roomName };
  }

  return { type: 'unknown' };

import type { TelegramMedia, TelegramMessage, TelegramUpdate } from './types';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function stringField(value: unknown): string | undefined {
  return typeof value === 'string' && value.trim() !== '' ? value : undefined;
}

function numberField(value: unknown): number | undefined {
  return typeof value === 'number' && Number.isFinite(value) ? value : undefined;
}

function chatIdField(value: unknown): number | string | undefined {
  return typeof value === 'number' || typeof value === 'string' ? value : undefined;
}

function mediaFrom(raw: Record<string, unknown>, key: 'document' | 'video' | 'voice'): TelegramMedia | null {
  const media = raw[key];
  if (!isRecord(media)) return null;
  const fileId = stringField(media.file_id);
  if (fileId == null) return null;

  return {
    fileId,
    fileName: stringField(media.file_name),
    mimeType: stringField(media.mime_type),
  };
}

function photoFrom(raw: Record<string, unknown>): TelegramMedia | null {
  const photos = raw.photo;
  if (!Array.isArray(photos) || photos.length === 0) return null;

  const largest = photos
    .filter(isRecord)
    .sort((a, b) => (numberField(b.file_size) ?? 0) - (numberField(a.file_size) ?? 0))[0];
  const fileId = stringField(largest?.file_id);
  return fileId == null ? null : { fileId, mimeType: 'image/jpeg' };
}

export function parseTelegramUpdate(value: unknown): TelegramUpdate | null {
  if (!isRecord(value)) return null;
  const messageRaw = value.message;
  if (messageRaw == null) return { updateId: numberField(value.update_id) };
  if (!isRecord(messageRaw) || !isRecord(messageRaw.chat) || !isRecord(messageRaw.from)) return null;

  const messageId = numberField(messageRaw.message_id);
  const chatId = chatIdField(messageRaw.chat.id);
  const fromId = chatIdField(messageRaw.from.id);
  if (messageId == null || chatId == null || fromId == null) return null;

  const replyRaw = isRecord(messageRaw.reply_to_message) ? messageRaw.reply_to_message : null;
  const base = {
    messageId,
    chatId,
    fromId,
    text: stringField(messageRaw.text),
    caption: stringField(messageRaw.caption),
    replyToMessageId: replyRaw == null ? undefined : numberField(replyRaw.message_id),
  };

  const photo = photoFrom(messageRaw);
  if (photo != null) return { updateId: numberField(value.update_id), message: { ...base, kind: 'photo', media: photo } };

  const document = mediaFrom(messageRaw, 'document');
  if (document != null) return { updateId: numberField(value.update_id), message: { ...base, kind: 'document', media: document } };

  const video = mediaFrom(messageRaw, 'video');
  if (video != null) return { updateId: numberField(value.update_id), message: { ...base, kind: 'video', media: video } };

  const voice = mediaFrom(messageRaw, 'voice');
  if (voice != null) return { updateId: numberField(value.update_id), message: { ...base, kind: 'voice', media: voice } };

  const message: TelegramMessage = { ...base, kind: base.text == null ? 'unsupported' : 'text' };
  return { updateId: numberField(value.update_id), message };
}

const TELEGRAM_API_BASE_URL = 'https://api.telegram.org';

export interface SentTelegramMessage {
  messageId: number;
}

export interface TelegramSendOptions {
  caption?: string;
  replyToMessageId?: number;
}

function messageIdFromPayload(payload: unknown): number {
  const result = payload as { result?: { message_id?: unknown } };
  const messageId = result.result?.message_id;
  if (typeof messageId !== 'number') {
    throw new Error('Telegram response missing message_id');
  }
  return messageId;
}

export class TelegramApi {
  constructor(
    private readonly botToken: string,
    private readonly fetchImpl: typeof fetch = fetch,
  ) {}

  private apiUrl(method: string): string {
    return `${TELEGRAM_API_BASE_URL}/bot${this.botToken}/${method}`;
  }

  fileUrl(filePath: string): string {
    return `${TELEGRAM_API_BASE_URL}/file/bot${this.botToken}/${filePath}`;
  }

  async request<T = unknown>(method: string, body: Record<string, unknown>): Promise<T> {
    const response = await this.fetchImpl(this.apiUrl(method), {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify(body),
    });
    const payload = await response.json().catch(() => null);
    if (!response.ok || (payload as { ok?: boolean } | null)?.ok === false) {
      throw new Error(`Telegram ${method} failed with HTTP ${response.status}`);
    }
    return payload as T;
  }

  async sendMessage(chatId: number | string, text: string, options: TelegramSendOptions = {}): Promise<SentTelegramMessage> {
    const body: Record<string, unknown> = { chat_id: chatId, text };
    if (options.replyToMessageId != null) {
      body.reply_to_message_id = options.replyToMessageId;
      body.allow_sending_without_reply = true;
    }
    const payload = await this.request('sendMessage', body);
    return { messageId: messageIdFromPayload(payload) };
  }

  async sendPhoto(chatId: number | string, photo: string, options: TelegramSendOptions = {}): Promise<SentTelegramMessage> {
    return this.sendMedia('sendPhoto', chatId, 'photo', photo, options);
  }

  async sendDocument(chatId: number | string, document: string, options: TelegramSendOptions = {}): Promise<SentTelegramMessage> {
    return this.sendMedia('sendDocument', chatId, 'document', document, options);
  }

  async sendVideo(chatId: number | string, video: string, options: TelegramSendOptions = {}): Promise<SentTelegramMessage> {
    return this.sendMedia('sendVideo', chatId, 'video', video, options);
  }

  async sendVoice(chatId: number | string, voice: string, options: TelegramSendOptions = {}): Promise<SentTelegramMessage> {
    return this.sendMedia('sendVoice', chatId, 'voice', voice, options);
  }

  private async sendMedia(method: string, chatId: number | string, field: string, value: string, options: TelegramSendOptions): Promise<SentTelegramMessage> {
    const body: Record<string, unknown> = { chat_id: chatId, [field]: value };
    if (options.caption != null && options.caption !== '') body.caption = options.caption;
    if (options.replyToMessageId != null) {
      body.reply_to_message_id = options.replyToMessageId;
      body.allow_sending_without_reply = true;
    }
    const payload = await this.request(method, body);
    return { messageId: messageIdFromPayload(payload) };
  }

  async getFilePath(fileId: string): Promise<string> {
    const payload = await this.request<{ result?: { file_path?: string } }>('getFile', { file_id: fileId });
    const filePath = payload.result?.file_path;
    if (typeof filePath !== 'string' || filePath.trim() === '') {
      throw new Error('Telegram getFile response missing file_path');
    }
    return filePath;
  }

  async downloadFile(filePath: string): Promise<Response> {
    const response = await this.fetchImpl(this.fileUrl(filePath));
    if (!response.ok) {
      throw new Error(`Telegram file download failed with HTTP ${response.status}`);
    }
    return response;
  }
}

import type { TelegramApi } from './api';

export interface DownloadedTelegramMedia {
  blob: Blob;
  name: string;
  type: string;
}

export function selectTelegramDownloadName(media: TelegramMedia, kind: TelegramMessageKind): string {
  if (media.fileName != null && media.fileName.trim() !== '') return media.fileName;
  const extension = kind === 'photo' ? 'jpg' : kind === 'video' ? 'mp4' : kind === 'voice' ? 'ogg' : 'bin';
  return `telegram-${kind}-${media.fileId}.${extension}`;
}

export async function downloadTelegramMedia(api: TelegramApi, media: TelegramMedia, kind: TelegramMessageKind = 'document'): Promise<DownloadedTelegramMedia> {
  const filePath = await api.getFilePath(media.fileId);
  const response = await api.downloadFile(filePath);
  const blob = await response.blob();
  const type = media.mimeType ?? response.headers.get('content-type') ?? 'application/octet-stream';

  return {
    blob,
    name: selectTelegramDownloadName(media, kind),
    type,
  };
}

export class HhhlApiClient {
  private readonly baseUrl: string;
  private readonly token: string;
  private readonly fetchImpl: typeof fetch;

  constructor(options: HhhlApiClientOptions) {
    this.baseUrl = options.baseUrl.replace(/\/$/, '');
    this.token = options.token;
    this.fetchImpl = options.fetchImpl ?? fetch;
  }

  async callEndpoint<T = unknown>(endpoint: string, params: Record<string, unknown> = {}): Promise<T> {
    const response = await this.fetchImpl(`${this.baseUrl}/${endpoint}`, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ ...params, i: this.token }),
    });
    const payload = await response.json().catch(() => null);
    if (!response.ok) {
      const error = payload as { error?: { code?: string; message?: string } } | null;
      throw new Error(error?.error?.message ?? `HHHL ${endpoint} failed with HTTP ${response.status}`);
    }
    return payload as T;
  }

  async uploadFile(formData: FormData): Promise<unknown> {
    formData.set('i', this.token);
    formData.set('force', 'true');
    const response = await this.fetchImpl(`${this.baseUrl}/drive/files/create`, {
      method: 'POST',
      body: formData,
    });
    const payload = await response.json().catch(() => null);
    if (!response.ok) {
      throw new Error(`HHHL drive upload failed with HTTP ${response.status}`);
    }
    return payload;

const HHHL_ORIGIN = 'https://dc.hhhl.cc';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function stringField(value: unknown): string | null {
  return typeof value === 'string' && value.trim() !== '' ? value.trim() : null;
}

function stringLike(value: unknown): string | null {
  if (typeof value === 'number' && Number.isFinite(value)) return String(value);
  return stringField(value);
}

function recordField(value: unknown): Record<string, unknown> | null {
  return isRecord(value) ? value : null;
}

function urlField(value: unknown): string | null {
  const raw = stringField(value);
  if (raw == null) return null;
  if (/^(?:https?:|blob:|data:)/.test(raw)) return raw;
  return raw.startsWith('/') ? `${HHHL_ORIGIN}${raw}` : raw;
}

function stringFrom(raw: Record<string, unknown>, keys: string[]): string | null {
  for (const key of keys) {
    const value = stringLike(raw[key]);
    if (value != null) return value;
  }
  return null;
}

function normalizeUser(value: unknown): HhhlUser | null {
  const raw = recordField(value);
  if (raw == null) return null;
  const id = stringFrom(raw, ['id', 'userId', 'accountId', 'username']) ?? '';
  const username = stringFrom(raw, ['username', 'acct', 'name']) ?? id;
  if (id === '' && username === '') return null;
  return { id: id === '' ? username : id, username, name: stringFrom(raw, ['name', 'displayName']) };
}

export function normalizeDriveFile(value: unknown): HhhlDriveFile | null {
  const raw = recordField(value);
  if (raw == null) return null;
  const id = stringFrom(raw, ['id', 'fileId', 'driveFileId']);
  const name = stringFrom(raw, ['name', 'fileName', 'filename']) ?? id;
  if (id == null || name == null) return null;
  return {
    id,
    name,
    type: stringFrom(raw, ['type', 'mimeType', 'contentType']),
    size: typeof raw.size === 'number' ? raw.size : null,
    url: urlField(raw.url ?? raw.webpublicUrl ?? raw.downloadUrl),
    thumbnailUrl: urlField(raw.thumbnailUrl ?? raw.thumbnail),
  };
}

function embedded(raw: Record<string, unknown>, key: 'reply' | 'quote', depth: number): HhhlChatMessage | null {
  if (depth > 1) return null;
  const source = recordField(raw[key]);
  return source == null ? null : normalizeChatMessage(source, depth + 1);
}

export function normalizeChatMessage(value: unknown, depth = 0): HhhlChatMessage {
  const raw = recordField(value) ?? {};
  return {
    id: stringFrom(raw, ['id', 'messageId', 'chatMessageId']) ?? '',
    roomId: stringFrom(raw, ['roomId', 'toRoomId']) ?? '',
    createdAt: stringFrom(raw, ['createdAt', 'created_at', 'created']) ?? new Date(0).toISOString(),
    text: stringFrom(raw, ['text', 'body', 'content', 'message']),
    user: normalizeUser(raw.user ?? raw.fromUser ?? raw.sender ?? raw.author),
    file: normalizeDriveFile(raw.file ?? raw.attachment ?? raw.driveFile),
    replyId: stringFrom(raw, ['replyId', 'replyToId', 'replyMessageId']),
    quoteId: stringFrom(raw, ['quoteId', 'quoteMessageId']),
    reply: embedded(raw, 'reply', depth),
    quote: embedded(raw, 'quote', depth),
  };
}

function normalizeMessages(value: unknown): HhhlChatMessage[] {
  const raw = recordField(value);
  const items = Array.isArray(value)
    ? value
    : Array.isArray(raw?.messages)
      ? raw.messages
      : Array.isArray(raw?.items)
        ? raw.items
        : [];

  return items.map((item) => normalizeChatMessage(item)).filter((message) => message.id !== '');
}

export function normalizeRoom(value: unknown): HhhlRoom {
  const raw = recordField(value) ?? {};
  const nested = recordField(raw.room) ?? raw;
  const id = stringFrom(nested, ['id', 'roomId']) ?? '';
  return { id, name: stringFrom(nested, ['name', 'title']) ?? id, description: stringFrom(nested, ['description']) };
}

export function normalizeUsers(value: unknown): HhhlUser[] {
  return Array.isArray(value) ? value.flatMap((item) => {
    const user = normalizeUser(recordField(item)?.user ?? recordField(item)?.member ?? item);
    return user == null ? [] : [user];
  }) : [];
}

export function createHhhlChatApi(client: HhhlEndpointCaller) {
  return {
    me: () => client.callEndpoint<unknown>('i', {}).then(normalizeUser),
    showRoom: (roomId: string) => client.callEndpoint<unknown>('chat/rooms/show', { roomId }).then(normalizeRoom),
    members: (roomId: string, params: PaginationParams = {}) => client.callEndpoint<unknown>('chat/rooms/members', { roomId, ...params }).then(normalizeUsers),
    roomTimeline: (roomId: string, params: PaginationParams = {}) =>
      client.callEndpoint<unknown>('chat/messages/room-timeline', { roomId, ...params }).then(normalizeMessages),
    createToRoom: (params: CreateRoomMessageParams) =>
      client.callEndpoint<unknown>('chat/messages/create-to-room', params).then(normalizeChatMessage),
  };
}

async function handleWebhook(request: Request, env: Env): Promise<Response> {
  const config = readConfig(env);
  if (!config.ok) return json({ ok: false, error: config.error }, { status: 500 });

  const raw = await readJson(request);
  const update = parseTelegramUpdate(raw);
  if (update == null) return json({ ok: false, error: 'invalid telegram update' }, { status: 400 });
  if (update.message == null) return json({ ok: true });

  if (String(update.message.fromId) !== config.value.allowedTelegramUserId) {
    return json({ ok: true });
  }

  const text = update.message.text;
  const command = text == null ? null : parseCommand(text);
  if (command?.type === 'help') {
    await new TelegramApi(config.value.botToken).sendMessage(update.message.chatId, commandHelpText);
    return json({ ok: true });
  }

  if (command?.type === 'invalid') {
    await new TelegramApi(config.value.botToken).sendMessage(update.message.chatId, command.reason);
    return json({ ok: true });
  }

  if (command?.type === 'unknown') {
    await new TelegramApi(config.value.botToken).sendMessage(update.message.chatId, '未知命令。发送 /help 查看帮助。');
    return json({ ok: true });
  }

  await new TelegramApi(config.value.botToken).sendMessage(update.message.chatId, '命令处理中断：该功能还没有接入。');
  return json({ ok: true });
}

    const url = new URL(request.url);

    }

    if (request.method === 'POST' && (url.pathname === '/' || url.pathname === '/webhook')) {

    }

async function validateBotMembership(config: AppConfig, roomId: string): Promise<{ ok: true; roomName: string } | { ok: false; message: string }> {

  const store = storeFor(context.env, context.config);

  if (command.type === 'bind') {
    const existing = await store.getBinding(context.telegramUserId);
    if (existing != null) {
      await context.telegram.sendMessage(context.chatId, `已经绑定：${existing.roomName} (${existing.roomId})。请先发送 /unbind。`);
      return;

    const validation = await validateBotMembership(context.config, command.roomId);
    if (!validation.ok) {
      await context.telegram.sendMessage(context.chatId, validation.message);
      return;

import type { HhhlChatMessage } from '../hhhl/types';
import type { KvStateStore } from '../state/kvStore';

export interface OutboundTelegramApi {
  sendMessage(chatId: number | string, text: string, options?: { replyToMessageId?: number }): Promise<{ messageId: number }>;
  sendPhoto(chatId: number | string, photo: string, options?: { caption?: string; replyToMessageId?: number }): Promise<{ messageId: number }>;
  sendDocument(chatId: number | string, document: string, options?: { caption?: string; replyToMessageId?: number }): Promise<{ messageId: number }>;
  sendVideo(chatId: number | string, video: string, options?: { caption?: string; replyToMessageId?: number }): Promise<{ messageId: number }>;
  sendVoice(chatId: number | string, voice: string, options?: { caption?: string; replyToMessageId?: number }): Promise<{ messageId: number }>;
}

export interface ForwardHhhlMessageOptions {
  message: HhhlChatMessage;
  telegramUserId: string;
  chatId: number | string;
  state: KvStateStore;
  telegram: OutboundTelegramApi;
  hhhlBotUserId: string;
  now?: () => string;
}

function authorName(message: HhhlChatMessage): string {
  return message.user?.username ?? message.user?.name ?? 'HHHL';
}

function textFor(message: HhhlChatMessage, quoteFallback: string | null): string {
  const body = message.text?.trim() ?? '';
  const prefix = `${authorName(message)}: `;
  const quote = quoteFallback == null ? '' : `引用：${quoteFallback}\n`;
  return `${quote}${prefix}${body}`.trim();
}

function fileMethod(type: string | null | undefined): 'photo' | 'video' | 'voice' | 'document' {
  if (type?.startsWith('image/')) return 'photo';
  if (type?.startsWith('video/')) return 'video';
  if (type === 'audio/ogg' || type === 'audio/opus' || type?.includes('voice')) return 'voice';
  return 'document';
}

async function replyTarget(state: KvStateStore, roomId: string, message: HhhlChatMessage): Promise<{ replyToMessageId?: number; quoteFallback: string | null }> {
  const targetId = message.replyId ?? message.quoteId;
  if (targetId == null) return { quoteFallback: null };
  const map = await state.getMessageMapByHhhl(roomId, targetId);
  if (map != null) return { replyToMessageId: map.telegramMessageId, quoteFallback: null };
  const embedded = message.reply ?? message.quote;
  const quoteFallback = embedded?.text == null ? targetId : `${authorName(embedded)}: ${embedded.text}`;
  return { quoteFallback };
}

export async function forwardHhhlMessageToTelegram(options: ForwardHhhlMessageOptions): Promise<void> {
  if (options.message.user?.id === options.hhhlBotUserId) return;
  const existing = await options.state.getMessageMapByHhhl(options.message.roomId, options.message.id);
  if (existing != null) return;

  const reply = await replyTarget(options.state, options.message.roomId, options.message);
  const file = options.message.file;
  const text = textFor(options.message, reply.quoteFallback);
  let sent: { messageId: number };

  if (file?.url != null) {
    const method = fileMethod(file.type);
    const caption = text === '' ? file.name : text;
    const sendOptions = { caption, replyToMessageId: reply.replyToMessageId };
    if (method === 'photo') sent = await options.telegram.sendPhoto(options.chatId, file.url, sendOptions);
    else if (method === 'video') sent = await options.telegram.sendVideo(options.chatId, file.url, sendOptions);
    else if (method === 'voice') sent = await options.telegram.sendVoice(options.chatId, file.url, sendOptions);
    else sent = await options.telegram.sendDocument(options.chatId, file.url, sendOptions);
  } else {
    sent = await options.telegram.sendMessage(options.chatId, text, { replyToMessageId: reply.replyToMessageId });
  }

  await options.state.putMessageMap({
    version: 1,
    telegramUserId: options.telegramUserId,
    telegramMessageId: sent.messageId,
    roomId: options.message.roomId,
    hhhlMessageId: options.message.id,
    createdAt: (options.now ?? (() => new Date().toISOString()))(),
  });
  await options.state.updateLastSeen(options.telegramUserId, options.message.id);
}

import { normalizeChatMessage } from './chatApi';
import type { HhhlChatMessage } from './types';

export type HhhlRealtimeEvent =
  | { type: 'message'; roomId: string; message: HhhlChatMessage }
  | { type: 'delete'; roomId: string; messageId: string }
  | { type: 'reaction'; roomId: string; messageId: string; reaction: string | null };

export function createStreamingUrl(hhhlOrigin: string, token: string): string {
  const origin = hhhlOrigin.replace(/^http:/, 'ws:').replace(/^https:/, 'wss:').replace(/\/$/, '');
  return `${origin}/streaming?i=${encodeURIComponent(token)}`;
}

export function createStreamConnectMessage(): unknown {
  return { type: 'connect', body: { channel: 'main', id: 'xbot-main', params: {}, pong: true } };
}

export function createRoomSubscribeMessage(roomId: string): unknown {
  return { type: 'ch', body: { id: `chatRoom:${roomId}`, type: 'connect', body: { roomId } } };
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

export function normalizeRealtimeEvent(value: unknown, subscribedRooms: Set<string>): HhhlRealtimeEvent | null {
  if (!isRecord(value) || value.type !== 'ch' || !isRecord(value.body)) return null;
  const eventType = value.body.type;
  const body = isRecord(value.body.body) ? value.body.body : null;
  if (body == null) return null;
  const rawMessage = body.message;
  const message = rawMessage == null ? null : normalizeChatMessage(rawMessage);
  const roomId = typeof body.roomId === 'string' ? body.roomId : message?.roomId;
  if (roomId == null || !subscribedRooms.has(roomId)) return null;

  if (eventType === 'message' && message != null) return { type: 'message', roomId, message };
  if (eventType === 'delete' && typeof body.messageId === 'string') return { type: 'delete', roomId, messageId: body.messageId };
  if (eventType === 'reaction' && typeof body.messageId === 'string') {
    return { type: 'reaction', roomId, messageId: body.messageId, reaction: typeof body.reaction === 'string' ? body.reaction : null };
  }
  return null;
}

import { createRoomSubscribeMessage, createStreamConnectMessage, createStreamingUrl, normalizeRealtimeEvent } from '../hhhl/realtime';
import type { HhhlChatMessage } from '../hhhl/types';
import type { KvStateStore } from '../state/kvStore';

interface WebSocketLike {
  onopen: (() => void) | null;
  onmessage: ((event: { data: string }) => void) | null;
  onclose: (() => void) | null;
  onerror: (() => void) | null;
  send(data: string): void;
  close(): void;
}

interface WebSocketCtor {
  new (url: string): WebSocketLike;
}

export interface BridgeRuntimeOptions {
  telegramUserId: string;
  chatId: number | string;
  hhhlOrigin: string;
  hhhlToken: string;
  hhhlBotUserId: string;
  state: KvStateStore;
  chatApi: { roomTimeline(roomId: string, params?: { limit?: number; sinceId?: string }): Promise<HhhlChatMessage[]> };
  outbound(message: HhhlChatMessage): Promise<void> | void;
  WebSocketImpl?: WebSocketCtor;
  setTimeoutImpl?: typeof setTimeout;
  now?: () => string;
  reconnectBaseDelayMs?: number;
  reconnectMaxDelayMs?: number;
  initialHistoryLimit?: number;
}

export class BridgeRuntime {
  private socket: WebSocketLike | null = null;
  private roomId: string | null = null;
  private reconnectAttempt = 0;
  private readonly subscribedRooms = new Set<string>();

  constructor(private readonly options: BridgeRuntimeOptions) {}

  async start(): Promise<void> {
    const binding = await this.options.state.getBinding(this.options.telegramUserId);
    if (binding == null) {
      await this.options.state.setStatus(this.options.telegramUserId, { version: 1, state: 'stopped', connectedAt: null, lastError: null, nextReconnectAt: null });
      return;
    }
    this.roomId = binding.roomId;
    this.subscribedRooms.clear();
    this.subscribedRooms.add(binding.roomId);
    await this.options.state.setStatus(this.options.telegramUserId, { version: 1, state: 'connecting', connectedAt: null, lastError: null, nextReconnectAt: null });
    this.openSocket();
  }

  async stop(): Promise<void> {
    this.socket?.close();
    this.socket = null;
    this.roomId = null;
    this.subscribedRooms.clear();
    await this.options.state.setStatus(this.options.telegramUserId, { version: 1, state: 'stopped', connectedAt: null, lastError: null, nextReconnectAt: null });
  }

  private openSocket(): void {
    const WebSocketImpl = this.options.WebSocketImpl ?? WebSocket;
    const socket = new WebSocketImpl(createStreamingUrl(this.options.hhhlOrigin, this.options.hhhlToken));
    this.socket = socket;

    socket.onopen = () => void this.handleOpen(socket);
    socket.onmessage = (event) => void this.handleMessage(event.data);
    socket.onerror = () => void this.handleSocketFailure('HHHL WebSocket error');
    socket.onclose = () => void this.handleSocketFailure('HHHL WebSocket closed');
  }

  private async handleOpen(socket: WebSocketLike): Promise<void> {
    this.reconnectAttempt = 0;
    await this.options.state.setStatus(this.options.telegramUserId, {
      version: 1,
      state: 'connected',
      connectedAt: (this.options.now ?? (() => new Date().toISOString()))(),
      lastError: null,
      nextReconnectAt: null,
    });
    socket.send(JSON.stringify(createStreamConnectMessage()));
    if (this.roomId != null) {
      socket.send(JSON.stringify(createRoomSubscribeMessage(this.roomId)));
      await this.backfill();
    }
  }

  private async backfill(): Promise<void> {
    const binding = await this.options.state.getBinding(this.options.telegramUserId);
    if (binding == null) return;
    const params = binding.lastSeenMessageId == null
      ? { limit: this.options.initialHistoryLimit ?? 30 }
      : { sinceId: binding.lastSeenMessageId, limit: this.options.initialHistoryLimit ?? 30 };
    const messages = await this.options.chatApi.roomTimeline(binding.roomId, params);
    for (const message of messages.slice().sort((a, b) => a.createdAt.localeCompare(b.createdAt))) {
      await this.options.outbound(message);
    }
  }

  private async handleMessage(data: string): Promise<void> {
    let parsed: unknown;
    try {
      parsed = JSON.parse(data) as unknown;
    } catch {
      return;
    }
    const event = normalizeRealtimeEvent(parsed, this.subscribedRooms);
    if (event?.type === 'message') {
      await this.options.outbound(event.message);
    }
  }

  async handleSocketFailure(reason: string): Promise<void> {
    if (this.roomId == null) return;
    this.reconnectAttempt += 1;
    const base = this.options.reconnectBaseDelayMs ?? 1000;
    const max = this.options.reconnectMaxDelayMs ?? 60000;
    const delay = Math.min(max, base * 2 ** Math.max(0, this.reconnectAttempt - 1));
    const nextReconnectAt = new Date(Date.now() + delay).toISOString();
    await this.options.state.setStatus(this.options.telegramUserId, {
      version: 1,
      state: 'backing_off',
      connectedAt: null,
      lastError: reason,
      nextReconnectAt,
    });
    const setTimeoutImpl = this.options.setTimeoutImpl ?? setTimeout;
    setTimeoutImpl(() => void this.start(), delay);
  }
}
